cmake_minimum_required(VERSION 3.24)
file(STRINGS "${CMAKE_CURRENT_SOURCE_DIR}/../VERSION" PROMPTCAD_VERSION LIMIT_COUNT 1)
project(PromptToCAD VERSION ${PROMPTCAD_VERSION} LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets Svg)
find_package(OpenCASCADE 7.8.1 EXACT REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms Visualization)
qt_add_executable(PromptToCAD WIN32 branding.qrc src/main.cpp src/WorkspaceWindow.cpp src/WorkspaceWindow.h src/MeshSource.cpp src/MeshSource.h src/CadView.cpp src/CadView.h src/RpcClient.cpp src/RpcClient.h)
target_sources(PromptToCAD PRIVATE src/WorkspaceControls.cpp src/WorkspaceControls.h src/WorkspacePanels.cpp src/WorkspaceEngineering.cpp src/WorkspaceTheme.cpp src/WorkspaceTheme.h src/StartupSplash.cpp src/StartupSplash.h src/WorkspaceSettings.cpp src/WorkspaceChat.cpp src/ConversationView.cpp src/ConversationView.h src/WorkspaceIcons.cpp src/WorkspaceIcons.h)
target_sources(PromptToCAD PRIVATE src/WorkspaceAttachments.cpp src/WorkspaceLayout.cpp src/WorkspaceProperties.cpp)
target_sources(PromptToCAD PRIVATE src/LoadingIndicator.cpp src/LoadingIndicator.h)
if(WIN32)
    enable_language(RC)
    configure_file(app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    target_sources(PromptToCAD PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
set_target_properties(PromptToCAD PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(PromptToCAD PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(PromptToCAD PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(PromptToCAD PRIVATE Qt6::Widgets Qt6::Svg)
foreach(toolkit TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKMesh TKMeshVS TKService TKV3d TKOpenGl)
    # SDK archives may retain build-machine paths in their transitive imported targets.
    find_library(CAD_${toolkit} NAMES ${toolkit} HINTS ${OpenCASCADE_LIBRARY_DIR} NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(PromptToCAD PRIVATE ${CAD_${toolkit}})
endforeach()
target_compile_definitions(PromptToCAD PRIVATE QT_NO_KEYWORDS NOMINMAX _USE_MATH_DEFINES PROMPTCAD_VERSION="${PROJECT_VERSION}")
if(MSVC)
    target_compile_options(PromptToCAD PRIVATE /W4 /utf-8 /EHsc)
    target_compile_options(PromptToCAD PRIVATE /experimental:deterministic "/pathmap:${CMAKE_CURRENT_SOURCE_DIR}/..=source")
    target_link_options(PromptToCAD PRIVATE /PDBALTPATH:PromptToCAD.pdb)
endif()
install(TARGETS PromptToCAD RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME native_workspace_smoke COMMAND "${CMAKE_CURRENT_SOURCE_DIR}/../.venv/Scripts/python.exe" -m tools.validate_ui)
set_tests_properties(native_workspace_smoke PROPERTIES TIMEOUT 540 WORKING_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/..")

option(PROMPTCAD_VIEWER_CHECK "Build the focused compound viewer check" OFF)
if(PROMPTCAD_VIEWER_CHECK)
    qt_add_executable(PromptCadViewerCheck tests/compound_viewer.cpp src/CadView.cpp src/CadView.h
        src/MeshSource.cpp src/MeshSource.h src/WorkspaceTheme.cpp src/WorkspaceTheme.h)
    set_target_properties(PromptCadViewerCheck PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
    target_include_directories(PromptCadViewerCheck PRIVATE src ${OpenCASCADE_INCLUDE_DIR})
    target_link_libraries(PromptCadViewerCheck PRIVATE Qt6::Widgets Qt6::Svg)
    foreach(toolkit TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKMesh TKMeshVS TKService TKV3d TKOpenGl)
        target_link_libraries(PromptCadViewerCheck PRIVATE "${CAD_${toolkit}}")
    endforeach()
    target_compile_definitions(PromptCadViewerCheck PRIVATE QT_NO_KEYWORDS NOMINMAX _USE_MATH_DEFINES)
    if(MSVC)
        target_compile_options(PromptCadViewerCheck PRIVATE /utf-8 /EHsc)
    endif()
endif()
