cmake_minimum_required(VERSION 3.24)
project(PromptToCAD VERSION 0.4.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_AUTORCC ON)
find_package(Qt6 6.5 REQUIRED COMPONENTS Widgets)
find_package(OpenCASCADE 7.8.1 EXACT REQUIRED COMPONENTS FoundationClasses ModelingData ModelingAlgorithms Visualization)
qt_add_executable(PromptToCAD WIN32 branding.qrc src/main.cpp src/WorkspaceWindow.cpp src/WorkspaceWindow.h src/MeshSource.cpp src/MeshSource.h src/CadView.cpp src/CadView.h src/RpcClient.cpp src/RpcClient.h)
if(WIN32)
    enable_language(RC)
    configure_file(app.rc.in "${CMAKE_CURRENT_BINARY_DIR}/app.rc" @ONLY)
    target_sources(PromptToCAD PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/app.rc")
endif()
set_target_properties(PromptToCAD PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
target_include_directories(PromptToCAD PRIVATE ${OpenCASCADE_INCLUDE_DIR})
target_link_directories(PromptToCAD PRIVATE ${OpenCASCADE_LIBRARY_DIR})
target_link_libraries(PromptToCAD PRIVATE Qt6::Widgets)
foreach(toolkit TKernel TKMath TKG2d TKG3d TKGeomBase TKBRep TKGeomAlgo TKTopAlgo TKMesh TKMeshVS TKService TKV3d TKOpenGl)
    # SDK archives may retain build-machine paths in their transitive imported targets.
    find_library(CAD_${toolkit} NAMES ${toolkit} HINTS ${OpenCASCADE_LIBRARY_DIR} NO_DEFAULT_PATH REQUIRED)
    target_link_libraries(PromptToCAD PRIVATE ${CAD_${toolkit}})
endforeach()
target_compile_definitions(PromptToCAD PRIVATE QT_NO_KEYWORDS NOMINMAX _USE_MATH_DEFINES CAD_SOURCE_ROOT="${CMAKE_CURRENT_SOURCE_DIR}/..")
if(MSVC)
    target_compile_options(PromptToCAD PRIVATE /W4 /utf-8 /EHsc)
endif()
install(TARGETS PromptToCAD RUNTIME DESTINATION bin)
enable_testing()
add_test(NAME native_workspace_smoke COMMAND PromptToCAD --workspace "${CMAKE_CURRENT_SOURCE_DIR}/.." --smoke-test --open "${CMAKE_CURRENT_SOURCE_DIR}/../examples/spindle-assembly.cad.json")
set_tests_properties(native_workspace_smoke PROPERTIES TIMEOUT 180)
